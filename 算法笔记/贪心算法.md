# 贪心算法

## 一、核心思想

贪心算法在每一步都选择当前状态下最优的选择，希望通过局部最优达到全局最优。

**适用条件**：
- 贪心选择性质：每一步的最优选择不依赖于未来的选择
- 最优子结构：问题的最优解包含子问题的最优解

## 二、区间问题

### 例题1：区间选点

```cpp
int intervalPoint(vector<vector<int>>& intervals) {
    sort(intervals.begin(), intervals.end(), [](auto& a, auto& b) {
        return a[1] < b[1];
    });
    
    int count = 0, last = -1;
    for (auto& interval : intervals) {
        if (interval[0] > last) {
            count++;
            last = interval[1];
        }
    }
    return count;
}
```

### 例题2：区间合并

```cpp
vector<vector<int>> mergeIntervals(vector<vector<int>>& intervals) {
    if (intervals.empty()) return {};
    
    sort(intervals.begin(), intervals.end(), [](auto& a, auto& b) {
        return a[0] < b[0];
    });
    
    vector<vector<int>> result;
    result.push_back(intervals[0]);
    
    for (int i = 1; i < intervals.size(); i++) {
        auto& last = result.back();
        if (intervals[i][0] <= last[1]) {
            last[1] = max(last[1], intervals[i][1]);
        } else {
            result.push_back(intervals[i]);
        }
    }
    return result;
}
```

int eraseOverlapIntervals(vector<vector<int>>& intervals) {

## 三、活动选择问题

```cpp
int activitySelection(vector<int>& start, vector<int>& end) {
    vector<pair<int, int>> activities;
    for (int i = 0; i < start.size(); i++) {
        activities.push_back({end[i], start[i]});
    }
    
    sort(activities.begin(), activities.end());
    
    int count = 0, last_end = -1;
    for (auto& activity : activities) {
        int s = activity.second;
        int e = activity.first;
        
        if (s >= last_end) {
            count++;
            last_end = e;
        }
    }
    return count;
}
```

## 四、哈夫曼编码

```cpp
int huffmanCoding(vector<int>& weights) {
    priority_queue<int, vector<int>, greater<int>> pq;
    
    for (int w : weights) {
        pq.push(w);
    }
    
    int total = 0;
    while (pq.size() > 1) {
        int a = pq.top(); pq.pop();
        int b = pq.top(); pq.pop();
        int sum = a + b;
        total += sum;
        pq.push(sum);
    }
    return total;
}
```

## 五、分数背包

```cpp
double fractionalKnapsack(vector<int>& weights, vector<int>& values, int capacity) {
    vector<pair<double, int>> items;
    for (int i = 0; i < weights.size(); i++) {
        items.push_back({(double)values[i] / weights[i], i});
    }
    
    sort(items.rbegin(), items.rend());
    
    double total_value = 0;
    int remaining = capacity;
    
    for (auto& item : items) {
        int idx = item.second;
        if (remaining >= weights[idx]) {
            total_value += values[idx];
            remaining -= weights[idx];
        } else {
            total_value += item.first * remaining;
            break;
        }
    }
    return total_value;
}
```

## 六、跳跃游戏

### 例题1：跳跃游戏

```cpp
bool canJump(vector<int>& nums) {
    int max_reach = 0;
    for (int i = 0; i < nums.size(); i++) {
        if (i > max_reach) return false;
        max_reach = max(max_reach, i + nums[i]);
        if (max_reach >= nums.size() - 1) return true;
    }
    return true;
}
```

### 例题2：跳跃游戏 II

```cpp
int jump(vector<int>& nums) {
    int jumps = 0, current_end = 0, farthest = 0;
    
    for (int i = 0; i < nums.size() - 1; i++) {
        farthest = max(farthest, i + nums[i]);
        if (i == current_end) {
            jumps++;
            current_end = farthest;
        }
    }
    return jumps;
}
```

## 七、分发糖果

```cpp
int candy(vector<int>& ratings) {
    int n = ratings.size();
    vector<int> candies(n, 1);
    
    for (int i = 1; i < n; i++) {
        if (ratings[i] > ratings[i - 1]) {
            candies[i] = candies[i - 1] + 1;
        }
    }
    
    for (int i = n - 2; i >= 0; i--) {
        if (ratings[i] > ratings[i + 1]) {
            candies[i] = max(candies[i], candies[i + 1] + 1);
        }
    }
    
    return accumulate(candies.begin(), candies.end(), 0);
}
```

## 八、贪心算法解题步骤

1. **确定贪心策略**：找出每一步的最优选择
2. **证明正确性**：验证贪心选择性质和最优子结构
3. **实现算法**：按策略逐步执行

## 九、常见贪心场景
