# 字符串

## 一、字符串匹配

### 暴力匹配

```cpp
int bruteForce(string s, string p) {
    int n = s.size(), m = p.size();
    for (int i = 0; i <= n - m; i++) {
        int j = 0;
        while (j < m && s[i + j] == p[j]) j++;
        if (j == m) return i;
    }
    return -1;
}
```

### KMP算法

```cpp
vector<int> buildNext(string p) {
    int m = p.size();
    vector<int> next(m, 0);
    
    for (int i = 1, j = 0; i < m; i++) {
        while (j > 0 && p[i] != p[j]) {
            j = next[j - 1];
        }
        if (p[i] == p[j]) {
            j++;
        }
        next[i] = j;
    }
    return next;
}

int kmp(string s, string p) {
    int n = s.size(), m = p.size();
    if (m == 0) return 0;
    
    vector<int> next = buildNext(p);
    
    for (int i = 0, j = 0; i < n; i++) {
        while (j > 0 && s[i] != p[j]) {
            j = next[j - 1];
        }
        if (s[i] == p[j]) {
            j++;
        }
        if (j == m) {
            return i - m + 1;
        }
    }
    return -1;
}
```

## 二、字符串哈希

### 前缀哈希

```cpp
typedef unsigned long long ull;

vector<ull> prefix_hash;
vector<ull> power;

void initHash(string s, ull base = 131) {
    int n = s.size();
    prefix_hash.resize(n + 1, 0);
    power.resize(n + 1, 1);
    
    for (int i = 0; i < n; i++) {
        prefix_hash[i + 1] = prefix_hash[i] * base + s[i];
        power[i + 1] = power[i] * base;
    }
}

ull getHash(int l, int r) {
    return prefix_hash[r] - prefix_hash[l - 1] * power[r - l + 1];
}
```

### 例题：最长回文子串

```cpp
string longestPalindrome(string s) {
    string t = "#";
    for (char c : s) {
        t += c;
        t += "#";
    }
    
    int n = t.size();
    vector<int> p(n, 0);
    int center = 0, right = 0;
    
    for (int i = 0; i < n; i++) {
        int mirror = 2 * center - i;
        if (i < right) {
            p[i] = min(right - i, p[mirror]);
        }
        
        int a = i + p[i] + 1;
        int b = i - p[i] - 1;
        while (a < n && b >= 0 && t[a] == t[b]) {
            p[i]++;
            a++;
            b--;
        }
        
        if (i + p[i] > right) {
            center = i;
            right = i + p[i];
        }
    }
    
    int max_len = 0, idx = 0;
    for (int i = 0; i < n; i++) {
        if (p[i] > max_len) {
            max_len = p[i];
            idx = i;
        }
    }
    
    int start = (idx - max_len) / 2;
    return s.substr(start, max_len);
}
```

## 三、AC自动机

### 核心思想
建立字典树，添加失败指针，实现多模式串匹配。

```cpp
struct TrieNode {
    TrieNode* children[26];
    TrieNode* fail;
    bool is_end;
    
    TrieNode() {
        memset(children, 0, sizeof(children));
        fail = nullptr;
        is_end = false;
    }
};

void buildAC(TrieNode* root) {
    queue<TrieNode*> q;
    root->fail = nullptr;
    q.push(root);
    
    while (!q.empty()) {
        TrieNode* u = q.front();
        q.pop();
        
        for (int i = 0; i < 26; i++) {
            if (u->children[i]) {
                TrieNode* v = u->children[i];
                TrieNode* f = u->fail;
                
                while (f && !f->children[i]) {
                    f = f->fail;
                }
                
                v->fail = f ? f->children[i] : root;
                q.push(v);
            }
        }
    }
}
```

## 四、后缀数组

### 倍增算法

```cpp
vector<int> sa, rank, height;

void buildSA(string s) {
    int n = s.size();
    sa.resize(n);
    rank.resize(n);
    
    for (int i = 0; i < n; i++) {
        sa[i] = i;
        rank[i] = s[i];
    }
    
    for (int k = 1; k < n; k <<= 1) {
        auto cmp = [&](int i, int j) {
            if (rank[i] != rank[j]) return rank[i] < rank[j];
            int ri = i + k < n ? rank[i + k] : -1;
            int rj = j + k < n ? rank[j + k] : -1;
            return ri < rj;
        };
        sort(sa.begin(), sa.end(), cmp);
        
        vector<int> new_rank(n);
        new_rank[sa[0]] = 0;
        for (int i = 1; i < n; i++) {
            new_rank[sa[i]] = new_rank[sa[i - 1]] + (cmp(sa[i - 1], sa[i]) ? 1 : 0);
        }
        rank = new_rank;
    }
}
```

## 五、常见字符串操作

### 反转字符串

```cpp
void reverseString(string& s) {
    int l = 0, r = s.size() - 1;
    while (l < r) {
        swap(s[l++], s[r--]);
    }
}
```

### 判断回文

```cpp
bool isPalindrome(string s) {
    int l = 0, r = s.size() - 1;
    while (l < r) {
        if (s[l++] != s[r--]) return false;
    }
    return true;
}
```

### 字符串分割

```cpp
vector<string> split(string s, char delimiter) {
    vector<string> result;
    string token;
    for (char c : s) {
        if (c == delimiter) {
            if (!token.empty()) {
                result.push_back(token);
                token.clear();
            }
        } else {
            token += c;
        }
    }
    if (!token.empty()) {
        result.push_back(token);
    }
    return result;
}
```

## 六、字符串算法对比

| 算法 | 用途 | 时间复杂度 | 空间复杂度 |
|------|------|-----------|-----------|
| 暴力匹配 | 单模式串匹配 | O(n*m) | O(1) |
| KMP | 单模式串匹配 | O(n+m) | O(m) |
| 字符串哈希 | 子串比较 | O(n)预处理，O(1)查询 | O(n) |
| AC自动机 | 多模式串匹配 | O(n+m+z) | O(m) |
| 后缀数组 | 多个子串问题 | O(n log n) | O(n) |
